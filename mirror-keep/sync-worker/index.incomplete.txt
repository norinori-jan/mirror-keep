function bytesToBase64(bytes) {
let s = '';
bytes.forEach(x => s += String.fromCharCode(x));

function base64ToBytes(b64) {
const bin = atob(b64);
const bytes = new Uint8Array(bin.length);
for (let i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
return bytes;
}

async function decryptOneField(enc) {
// 新形式（envelope: iv/ciphertextがbase64文字列）と
// 旧形式（iv/dataが生バイト、ivを使い回していたデータ）の両方に対応
const isEnvelope = typeof enc.iv === 'string';
const iv = isEnvelope ? base64ToBytes(enc.iv) : enc.iv;
const data = isEnvelope ? base64ToBytes(enc.ciphertext) : enc.data;
const buf = await crypto.subtle.decrypt({ name: 'AES-GCM', iv }, sessionKey, data);
return new TextDecoder().decode(buf);
}

const fieldsJson = await decryptOneField(item.encFields);
const fields = JSON.parse(fieldsJson);

const out = { body, fields };
decryptedCache.set(item.id, out);
return out;
} catch (_) {

}

async function encryptField(plaintext) {
const meta = getPinMeta() || {};
const iv = crypto.getRandomValues(new Uint8Array(12));
const buf = new TextEncoder().encode(plaintext);
const cipherBuf = await crypto.subtle.encrypt(
{ name: 'AES-GCM', iv },
sessionKey,
buf
);
return {
v: 1,
keySource: 'pin',
alg: 'AES-GCM',
kdf: 'PBKDF2',
iterations: meta.iterations || 150000,
salt: meta.salt,
iv: bytesToBase64(iv),
ciphertext: bytesToBase64(new Uint8Array(cipherBuf))
};
}

function escapeHtml(str) {
return (str || '')
.replace(/&/g, '&amp;')
.replace(/</g, '&lt;')
.replace(/>/g, '&gt;')
.replace(/"/g, '&quot;')
.replace(/'/g, '&#39;');
}

function bigrams(str) {
const s = (str || '').toLowerCase().replace(/\s+/g, '');
const r = new Set();
for (let i = 0; i < s.length - 1; i++) r.add(s.slice(i, i + 2));
if (s.length === 1) r.add(s);
return r;
}

function buildSearchText(item) {
const cached = decryptedCache.get(item.id);
const body = cached ? cached.body : (item.body || '');
const fieldsText = (cached ? cached.fields : (item.fields || []))
.map(f => `${f.key} ${f.value}`)
.join(' ');
return [
item.title || '',
body,
(item.tags || []).join(' '),
fieldsText,
item.url || ''
].join(' ').toLowerCase();
}

function searchItems(items, query) {
if (!query.trim()) return items;
const q = query.trim().toLowerCase();
const qBi = bigrams(q);

return items.filter(item => {
const text = buildSearchText(item);
if (text.includes(q)) return true;

if (q.length >= 2) {
const tBi = bigrams(text);
let m = 0;
for (const b of qBi) if (tBi.has(b)) m++;
return m / qBi.size >= 0.5;
}
return false;
}).sort((a, b) => {
const aT = (a.title || '').toLowerCase().includes(q) ? 1 : 0;
const bT = (b.title || '').toLowerCase().includes(q) ? 1 : 0;
if (bT !== aT) return bT - aT;
return (b.updatedAt || 0) - (a.updatedAt || 0);
});
}

function highlightText(text, query) {
if (!query.trim()) return escapeHtml(text);
const q = escapeHtml(query.trim());
const escaped = escapeHtml(text);
const re = new RegExp(q.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'), 'gi');
return escaped.replace(re, m => `<mark>${m}</mark>`);
}

async function saveItem() {
const title = document.getElementById('fTitle').value.trim();
if (!title) {
showToast('タイトルを入力してください');
return;
}

const url = document.getElementById('inputUrl').value.trim();
const body = document.getElementById('fBody').innerHTML;
const fields = draftFields.filter(f => f.key.trim() || f.value.trim());
const now = Date.now();

let item = {
id: editingId || crypto.randomUUID(),
title,
tags: [...draftTags],
url,
sensitive: draftSensitive,
createdAt: editingId ? allItems.find(i => i.id === editingId).createdAt : now,
updatedAt: now
};

if (draftSensitive) {
if (!sessionKey) {
const ok = await ensureUnlocked();
if (!ok) return;
}
const enc = await encryptToItemFields(body, fields);
item.encBody = enc.encBody;
item.encFields = enc.encFields;
item.body = '';
item.fields = [];
} else {
item.body = body;
item.fields = fields;
item.encBody = null;
item.encFields = null;
}

await dbPut(item);
window.QuickRefBridge?.onItemSaved(item);

for (const attId of pendingDeleteAttIds) await dbDeleteAttachment(attId);
pendingDeleteAttIds = [];

await saveImageAttachments(item.id);
await saveAudioAttachments(item.id);
await saveVideoAttachments(item.id);

await reload();
closeSheet();
showToast('保存しました');
void syncCloudNow(false);
}

function tryImportFromIllustStudio() {
try {
const raw = localStorage.getItem('is_transfer');
if (!raw) {
alert('illust-studioからの転送データが見つかりません。');
return;
}
const data = JSON.parse(raw);
const src = data.image || data.dataUrl || data.src || null;
if (!src) {
alert('画像データが取得できませんでした。');
return;
}
fetch(src).then(r => r.blob()).then(blob => {
addBlobToDraft(blob, data.title ? `${data.title}.png` : 'illust-studio.png');
localStorage.removeItem('is_transfer');
});
} catch (e) {
alert('is_transfer読み込み失敗: ' + e.message);
}
}

function checkImportHash() {
if (!location.hash.startsWith('#import=')) return;
try {
const payload = JSON.parse(decodeURIComponent(escape(atob(location.hash.slice(8)))));
const items = Array.isArray(payload.items) ? payload.items : payload;
if (Array.isArray(items)) importItemsFromPayload(items, payload.source || 'unknown');
} catch (e) {
console.warn(e);
} finally {
history.replaceState(null, '', location.pathname + location.search);
}
}

function checkIncomingTransfer() {
try {
const raw = localStorage.getItem(LS.TRANSFER);
if (!raw) return;
const payload = JSON.parse(raw);
if (payload.source === 'quick-ref' || !Array.isArray(payload.items)) return;
if (confirm(`${payload.source || '他アプリ'}から${payload.items.length}件のデータがあります。インポートしますか？`)) {
importItemsFromPayload(payload.items, payload.source || 'transfer');
localStorage.removeItem(LS.TRANSFER);
}
} catch (_) {}
}
